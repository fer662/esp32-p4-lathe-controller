cmake_minimum_required(VERSION 3.16)
option(LATHE_BENCH_ONLY "Lock motor enables and include disconnected-bench diagnostics" OFF)
option(LATHE_OTA_REQUIRE_PAIRING "Require a temporary pairing key for OTA uploads" OFF)
option(LATHE_LEGACY_OTA_BRIDGE "Use the old image identity for the one-time rename transition" OFF)
set(P4_BENCH_ONLY ${LATHE_BENCH_ONLY} CACHE BOOL "Driver enable policy" FORCE)
set(GRBLHAL_P4_CUSTOM_SERVICES ON CACHE BOOL "Application-owned services" FORCE)
set(GRBLHAL_CORE_ROOT "${CMAKE_CURRENT_LIST_DIR}/dependencies" CACHE PATH "Parent of the pinned grbl core" FORCE)
set(GRBLHAL_P4_CONFIG_HEADER "${CMAKE_CURRENT_LIST_DIR}/main/machine.h" CACHE FILEPATH "Core and driver configuration" FORCE)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/dependencies/esp32/p4/components/grblhal_p4")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(LATHE_LEGACY_OTA_BRIDGE)
    # Existing firmware accepts this identity only; the bridge accepts both names.
    project(h5_grblhal_p4 VERSION 0.4.0)
else()
    project(esp32-p4-lathe-controller VERSION 0.4.0)
endif()
